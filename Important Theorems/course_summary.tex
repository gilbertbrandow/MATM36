\section*{Metric Spaces}

\begin{definition}
Let $X$ be a topological space and $A \subseteq X$. A point $x \in X$ is called an \emph{adherent point} of $A$ if every open neighbourhood $U$ of $x$ satisfies
\[
U \cap A \neq \varnothing.
\]
\end{definition}

In the special case where $X$ is a metric space with metric $d$, this is equivalent to requiring that for every $r>0$,
\[
B_r(x) \cap A \neq \varnothing.
\]

\begin{definition}
Let $X$ be a topological space and $A \subseteq X$. The set $A$ is called \emph{dense} in $X$ if every nonempty open set $U \subseteq X$ satisfies
\[
U \cap A \neq \varnothing.
\]
Equivalently, $A$ is dense in $X$ if $\overline{A} = X$.
\end{definition}

\begin{definition}
Let $X$ be a topological space and $A \subseteq X$. 
The set $A$ is called \emph{nowhere dense} in $X$ if
\[
\operatorname{int}(\overline{A}) = \varnothing.
\]
\end{definition}

\begin{definition}
A topological space $X$ is \emph{separable} if: $\exists D \subseteq X : D$ countable and dense.
\end{definition}

\subsection*{Baire Category Theorem}

\begin{theorem}[Baire Category Theorem]
Let $(X,d)$ be a complete metric space and let $\{U_n\}_{n=1}^{\infty}$ be a sequence of dense open subsets of $X$. Then \(\bigcap_{n=1}^{\infty} U_n\) is dense in $X$.
\end{theorem}

\begin{proof}
Fix a nonempty open ball $B(x,\varepsilon)\subset X$.

Since $U_1$ is dense and open, choose $y_1\in U_1\cap B(x,\varepsilon)$ and $r_1>0$ such that
\[
\overline{B}(y_1,r_1)\subset U_1\cap B(x,\varepsilon),\qquad r_1<1.
\]
Inductively, having chosen $y_n,r_n$, use that $U_{n+1}$ is dense and open to pick
$y_{n+1}\in U_{n+1}\cap B(y_n,r_n)$ and $r_{n+1}>0$ such that
\[
\overline{B}(y_{n+1},r_{n+1})\subset U_{n+1}\cap B(y_n,r_n),\qquad r_{n+1}<\frac1{n+1}.
\]
Then the closed balls are nested:
\[
\overline{B}(y_{n+1},r_{n+1})\subset \overline{B}(y_n,r_n)\subset \cdots \subset \overline{B}(y_1,r_1)\subset B(x,\varepsilon),
\]
and for $m>n$ we have $y_m\in B(y_n,r_n)$, hence $d(y_m,y_n)<r_n\to 0$. Thus $(y_n)$ is Cauchy, so by completeness $y_n\to y\in X$.

For each fixed $n$, all tails of the sequence lie in $\overline{B}(y_n,r_n)$, so the limit satisfies
$y\in \overline{B}(y_n,r_n)\subset U_n$. Hence $y\in \bigcap_{n=1}^\infty U_n$, and also
$y\in \overline{B}(y_1,r_1)\subset B(x,\varepsilon)$. Therefore
\[
B(x,\varepsilon)\cap\bigcap_{n=1}^\infty U_n\neq\varnothing,
\]
so the intersection is dense.
\end{proof}


\begin{remark}
From the definitions it follows that a set $B \subseteq X$ satisfies 
\(\operatorname{int}(B)=\varnothing \Longleftrightarrow X\setminus B \text{ is dense in } X.\)
In particular, if $A$ is nowhere dense, then
\( X\setminus \overline{A}\) 
is open and dense in $X$.
\end{remark}

\begin{corollary}
Let $(X,d)$ be a complete metric space and let $\{F_n\}_{n=1}^{\infty}$ be a sequence of nowhere dense subsets of $X$. Then
\[
\operatorname{int}\!\left(\bigcup_{n=1}^{\infty} F_n\right)=\varnothing.
\]
\end{corollary}

\begin{proof}
Since $F_n$ is nowhere dense, $U_n:=X\setminus\overline{F_n}$ is open and dense; by Baire, $\bigcap_{n=1}^\infty U_n = \bigcup_{n=1}^\infty \overline{F_n}$ is dense, hence nonempty in every nonempty open set.
\end{proof}

\subsection*{Finite Products of Metric Spaces}

\begin{definition}
Let $(X_k,d_k)$, $k=1,\dots,n$, be metric spaces and set
\[
X = X_1 \times \cdots \times X_n.
\]
A metric $d$ on $X$ is said to be a \emph{product metric} if convergence in $(X,d)$ is coordinatewise,
\[
x^{(m)} = (x_1^{(m)},\dots,x_n^{(m)}) \to x=(x_1,\dots,x_n)
\]
in $X$ if and only if for each $k=1,\dots,n$,
\[
x_k^{(m)} \to x_k \quad \text{in } X_k.
\]
\end{definition}

\begin{theorem}
Suppose $d$ is a metric on $X=X_1\times\cdots\times X_n$ satisfying the above coordinatewise convergence property. Then the open sets in $(X,d)$ are precisely the unions of product sets
\[
U_1 \times \cdots \times U_n,
\]
where each $U_k$ is open in $X_k$.
\end{theorem}

\begin{proof}
First, if $E_k \subset X_k$ is closed for each $k$, then
\[
E_1 \times \cdots \times E_n
\]
is closed in $X$. Indeed, if $x^{(m)} \in E_1 \times \cdots \times E_n$ and $x^{(m)} \to x$ in $X$, then by coordinatewise convergence,
$x_k^{(m)} \to x_k$ in $X_k$. Since each $E_k$ is closed, $x_k \in E_k$, hence
$x \in E_1 \times \cdots \times E_n$.

Now let $U_k \subset X_k$ be open. Then
\[
V^{(k)} := X_1 \times \cdots \times X_{k-1} \times U_k \times X_{k+1} \times \cdots \times X_n
\]
is open in $X$, since its complement is of the form
\[
X_1 \times \cdots \times (X_k \setminus U_k) \times \cdots \times X_n,
\]
which is closed by the previous remark. Hence
\[
U_1 \times \cdots \times U_n
=
\bigcap_{k=1}^n V^{(k)}
\]
is open in $X$, and any union of such products is open.

Conversely, let $U \subset X$ be open and $x=(x_1,\dots,x_n)\in U$.
We show there exist open sets $U_k \subset X_k$ such that
\[
x \in U_1 \times \cdots \times U_n \subset U.
\]
Suppose not. Then for each $m\in\mathbb{N}$,
\[
B_{X_1}(x_1,1/m) \times \cdots \times B_{X_n}(x_n,1/m)
\not\subset U.
\]
Thus there exists $x^{(m)} \notin U$ with
\[
d_k(x_k^{(m)},x_k) < \frac1m, \qquad k=1,\dots,n.
\]
Then $x_k^{(m)} \to x_k$ for each $k$, hence $x^{(m)} \to x$ in $X$.
Since $X\setminus U$ is closed and each $x^{(m)} \in X\setminus U$,
we conclude $x \in X\setminus U$, a contradiction.
Therefore such open sets $U_k$ exist, and $U$ is a union of product sets.
\end{proof}

\begin{theorem}[Heine--Borel Theorem]
For a subspace $E \subseteq \mathbb{R}^n$, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $E$ is compact;
\item every sequence in $E$ has a convergent subsequence with limit in $E$;
\item $E$ is closed and bounded.
\end{enumerate}
\end{theorem}

\begin{proof}
By the general compactness theorem for metric spaces, for any metric space the following are equivalent:
compactness, sequential compactness, and completeness together with total boundedness.

Now let $E \subseteq \mathbb{R}^n$.

If $E$ is closed and bounded, then $E$ is complete (since a subspace of $\mathbb{R}^n$ is complete if and only if it is closed) and totally bounded (since a subset of $\mathbb{R}^n$ is totally bounded if and only if it is bounded). Hence $E$ is compact, and therefore every sequence in $E$ has a convergent subsequence.

Conversely, if $E$ is compact, then by the same general theorem $E$ is complete and totally bounded. Since $E$ is a subspace of $\mathbb{R}^n$, completeness implies that $E$ is closed, and total boundedness implies that $E$ is bounded. Thus $E$ is closed and bounded.

Therefore (i), (ii), and (iii) are equivalent.
\end{proof}

\begin{theorem}[Uniform Continuity Theorem]
Let $(X,d)$ and $(Y,\rho)$ be metric spaces, and suppose that $X$ is compact.
Then every continuous function $f:X\to Y$ is uniformly continuous.
\end{theorem}

\begin{proof}
Suppose $f$ is not uniformly continuous. Then there exists $\varepsilon>0$ such that for every $m\in\mathbb N$ there are points $x_m,z_m\in X$ with
\[
d(x_m,z_m)<\frac1m
\qquad\text{but}\qquad
\rho\bigl(f(x_m),f(z_m)\bigr)\ge \varepsilon.
\]
Since $X$ is compact, the sequence $(x_m)$ has a convergent subsequence; passing to that subsequence, we may assume
\[
x_m\to x\in X.
\]
Because $d(x_m,z_m)\to 0$, also
\[
z_m\to x.
\]
By continuity of $f$ at $x$,
\[
f(x_m)\to f(x)
\qquad\text{and}\qquad
f(z_m)\to f(x).
\]
Hence
\[
\rho\bigl(f(x_m),f(z_m)\bigr)
\le
\rho\bigl(f(x_m),f(x)\bigr)+\rho\bigl(f(x),f(z_m)\bigr)\to 0,
\]
which contradicts
\[
\rho\bigl(f(x_m),f(z_m)\bigr)\ge \varepsilon.
\]
Therefore $f$ is uniformly continuous.
\end{proof}

\begin{theorem}[Continuity and boundedness of linear operators]
Let $X$ and $Y$ be normed spaces and let $T:X\to Y$ be linear. The following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $T$ is continuous;
\item $T$ is continuous at $0$;
\item $T$ is bounded, i.e.\ there exists $C>0$ such that
\[
\|T(x)\|\le C\|x\| \quad \text{for all } x\in X.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
(i)$\Rightarrow$(ii) is immediate.

(ii)$\Rightarrow$(iii): Since $T$ is continuous at $0$, there exists $\delta>0$ such that
\[
\|x\|\le \delta \;\Rightarrow\; \|T(x)\|\le 1.
\]
For arbitrary $x\neq 0$, set $y=\frac{\delta}{\|x\|}x$, so $\|y\|=\delta$ and hence
\[
\|T(y)\|\le 1.
\]
By linearity,
\[
\|T(x)\| = \frac{\|x\|}{\delta}\|T(y)\|\le \frac{1}{\delta}\|x\|.
\]
Thus $T$ is bounded.

(iii)$\Rightarrow$(i): If $\|T(x)\|\le C\|x\|$, then for all $x,x_0\in X$,
\[
\|T(x)-T(x_0)\| = \|T(x-x_0)\|\le C\|x-x_0\|,
\]
so $T$ is Lipschitz, hence continuous.
\end{proof}

\begin{theorem}[Principle of Uniform Boundedness]
Let $X$ and $Y$ be Banach spaces and let $\{T_\alpha\}_{\alpha\in A}$ be a family of continuous linear operators $T_\alpha:X\to Y$. Suppose that for each $x\in X$,
\[
\sup_{\alpha\in A}\|T_\alpha x\| < \infty.
\]
Then
\[
\sup_{\alpha\in A}\|T_\alpha\| < \infty.
\]
\end{theorem}

\begin{proof}
For $n\in\mathbb N$, define
\[
E_n := \left\{ x\in X : \sup_{\alpha\in A}\|T_\alpha x\|\le n \right\}.
\]
Then each $E_n$ is closed, $E_n\subseteq E_{n+1}$, and by assumption
\[
X=\bigcup_{n=1}^\infty E_n.
\]
By the Baire Category Theorem, some $E_m$ has nonempty interior. Thus there exist $x_0\in X$ and $\varepsilon>0$ such that
\[
B(x_0,\varepsilon)\subseteq E_m.
\]

Let $c:=\sup_{\alpha\in A}\|T_\alpha x_0\|<\infty$. If $\|x\|<1$, then $x_0+\varepsilon x\in E_m$, hence
\[
\|T_\alpha(x_0+\varepsilon x)\|\le m.
\]
Thus
\[
\varepsilon\|T_\alpha x\|
=
\|T_\alpha(x_0+\varepsilon x)-T_\alpha(x_0)\|
\le m + c.
\]
Therefore
\[
\|T_\alpha x\|\le \frac{m+c}{\varepsilon}
\quad \text{for all } \|x\|<1.
\]
Hence $\|T_\alpha\|\le \frac{m+c}{\varepsilon}$ for all $\alpha\in A$, and so
\[
\sup_{\alpha\in A}\|T_\alpha\|<\infty.
\]
\end{proof}

\section*{Topologies}

\begin{definition}
Let $X$ be a set. A \emph{topology} on $X$ is a collection $\mathcal{T} \subseteq \mathcal{P}(X)$ of subsets of $X$ such that

\begin{enumerate}[label=(\roman*)]
\item $\varnothing \in \mathcal{T}$ and $X \in \mathcal{T}$,

\item if $\{U_\alpha\}_{\alpha \in I} \subseteq \mathcal{T}$, then
\[
\bigcup_{\alpha \in I} U_\alpha \in \mathcal{T},
\]

\item if $U_1,\dots,U_n \in \mathcal{T}$, then
\[
\bigcap_{i=1}^{n} U_i \in \mathcal{T}.
\]
\end{enumerate}

The pair $(X,\mathcal{T})$ is called a \emph{topological space}. The sets $U \in \mathcal{T}$ are called \emph{open sets}.

A subset $F \subseteq X$ is called \emph{closed} if its complement is open.
\end{definition}

\subsection*{Separation Axioms}

\begin{definition}[$T_0$ space]
A topological space $(X,\mathcal{T})$ is called a \emph{$T_0$-space} if for every pair of distinct points $x,y \in X$, there exists an open set $U \in \mathcal{T}$ such that
\[
x \in U,\; y \notin U
\quad \text{or} \quad
y \in U,\; x \notin U.
\]
\end{definition}

\begin{definition}[$T_1$ space]
A topological space $(X,\mathcal{T})$ is called a \emph{$T_1$-space} if for every pair of distinct points $x,y \in X$, there exists an open set $U \in \mathcal{T}$ such that
\[
y \in U \quad \text{and} \quad x \notin U.
\]
Equivalently, $(X,\mathcal{T})$ is $T_1$ if and only if every singleton set $\{x\}$ is closed in $X$.
\end{definition}

\begin{definition}[$T_2$ space / Hausdorff]
A topological space $(X,\mathcal{T})$ is called a \emph{$T_2$-space}, or \emph{Hausdorff}, if for every pair of distinct points $x,y \in X$, there exist open sets $U,V \in \mathcal{T}$ such that
\[
x \in U, \qquad y \in V, \qquad U \cap V = \varnothing.
\]
\end{definition}

\begin{definition}[Regular space]
A topological space $(X,\mathcal{T})$ is called \emph{regular} if for every closed set $F \subseteq X$ and every point $x \in X \setminus F$, there exist open sets $U,V \in \mathcal{T}$ such that
\[
x \in U, \qquad F \subseteq V, \qquad U \cap V = \varnothing.
\]
\end{definition}

\begin{definition}[$T_3$ space]
A topological space $(X,\mathcal{T})$ is called a \emph{$T_3$-space} if it is both $T_1$ and regular.
\end{definition}

\begin{definition}[Normal space]
A topological space $(X,\mathcal{T})$ is called \emph{normal} if for any two disjoint closed sets $F,G \subseteq X$, there exist open sets $U,V \in \mathcal{T}$ such that
\[
F \subseteq U, \qquad G \subseteq V, \qquad U \cap V = \varnothing.
\]
\end{definition}

\begin{definition}[$T_4$ space]
A topological space $(X,\mathcal{T})$ is called a \emph{$T_4$-space} if it is both $T_1$ and normal.
\end{definition}

\begin{theorem}
Every compact Hausdorff space is normal. In particular, every compact Hausdorff space is a $T_4$-space.
\end{theorem}

\begin{proof}[Proof]
Let $F,G \subseteq X$ be disjoint closed sets. Since $X$ is Hausdorff, for each $x \in F$ and $y \in G$ there exist disjoint open sets $U_{x,y}$ and $V_{x,y}$ such that
\[
x \in U_{x,y}, \qquad y \in V_{x,y}, \qquad U_{x,y} \cap V_{x,y} = \varnothing.
\]

For fixed $x \in F$, the sets $\{V_{x,y}\}_{y \in G}$ form an open cover of $G$. Since $G$ is closed in the compact space $X$, it is compact, so we may choose a finite subcover
\[
G \subseteq V_{x,y_1} \cup \cdots \cup V_{x,y_n}.
\]

Let
\[
U_x := U_{x,y_1} \cap \cdots \cap U_{x,y_n}, 
\qquad
V_x := V_{x,y_1} \cup \cdots \cup V_{x,y_n}.
\]
Then $x \in U_x$, $G \subseteq V_x$, and $U_x \cap V_x = \varnothing$.

The family $\{U_x\}_{x \in F}$ is an open cover of $F$. Since $F$ is compact, choose a finite subcover
\[
F \subseteq U_{x_1} \cup \cdots \cup U_{x_m}.
\]

Define
\[
U := U_{x_1} \cup \cdots \cup U_{x_m},
\qquad
V := V_{x_1} \cap \cdots \cap V_{x_m}.
\]

Then $U$ and $V$ are open, $F \subseteq U$, $G \subseteq V$, and $U \cap V = \varnothing$. Hence $X$ is normal.
\end{proof}

\begin{remark}[Continuity criterion in metric spaces]
Let $(X,\mathcal{T})$ be a topological space and $(Y,d)$ a metric space. 
A function $f:X\to Y$ is continuous at $x\in X$ if and only if for every $\varepsilon>0$ there exists a neighbourhood $U$ of $x$ such that
\[
d(f(y),f(x)) < \varepsilon \quad \text{for all } y\in U.
\]
Equivalently, $f(U) \subseteq B(f(x),\varepsilon)$.
\end{remark}

\begin{theorem}[Uniform limit preserves continuity]
Let $(X,\mathcal{T})$ be a topological space and $(Y,d)$ a metric space. 
Suppose $(f_n)$ is a sequence of functions $f_n:X\to Y$ converging uniformly to $f:X\to Y$. 
If each $f_n$ is continuous at $x\in X$, then $f$ is continuous at $x$.
\end{theorem}

\begin{proof}
We first recall the following characterization of continuity at a point:
$f$ is continuous at $x$ if and only if for every $\varepsilon>0$ there exists a neighbourhood $U$ of $x$ such that
\[
d(f(y),f(x))<\varepsilon \quad \text{for all } y\in U.
\]

Now let $\varepsilon>0$. By uniform convergence, there exists $N\in\mathbb{N}$ such that
\[
d(f_N(y),f(y))<\frac{\varepsilon}{3} \quad \text{for all } y\in X.
\]
Since $f_N$ is continuous at $x$, there exists a neighbourhood $U$ of $x$ such that
\[
d(f_N(y),f_N(x))<\frac{\varepsilon}{3} \quad \text{for all } y\in U.
\]
Then for all $y\in U$,
\[
d(f(y),f(x))
\le d(f(y),f_N(y)) + d(f_N(y),f_N(x)) + d(f_N(x),f(x))
< \frac{\varepsilon}{3} + \frac{\varepsilon}{3} + \frac{\varepsilon}{3}
= \varepsilon.
\]
By the above characterization, $f$ is continuous at $x$.
\end{proof}


\begin{theorem}
Every metric space $(X,d)$ is a $T_4$-space.
\end{theorem}

\begin{proof}
First, for each $x\in X$,
\[
\{x\} = \bigcap_{r>0} \overline{B(x,r)},
\]
so singletons are closed and $X$ is $T_1$.

Let $E,F \subseteq X$ be disjoint closed sets. For each $x\in E$, there exists $r(x)>0$ such that
\[
B(x,r(x)) \cap F = \varnothing.
\]
Similarly, for each $y\in F$, there exists $r(y)>0$ such that
\[
B(y,r(y)) \cap E = \varnothing.
\]

Define
\[
U := \bigcup_{x\in E} B\!\left(x,\tfrac{r(x)}{2}\right),
\quad
V := \bigcup_{y\in F} B\!\left(y,\tfrac{r(y)}{2}\right).
\]
Then $U,V$ are open, $E\subseteq U$, $F\subseteq V$.

Suppose $z\in U\cap V$. Then there exist $x\in E$, $y\in F$ such that
\[
d(x,z) < \tfrac{r(x)}{2}, \quad d(y,z) < \tfrac{r(y)}{2}.
\]
Hence
\[
d(x,y) \le d(x,z)+d(z,y) < \tfrac{r(x)+r(y)}{2} \le \max\{r(x),r(y)\}.
\]
Thus either $d(x,y)<r(x)$ or $d(x,y)<r(y)$, contradicting the choice of $r(x)$ or $r(y)$.
Therefore $U\cap V=\varnothing$, and $X$ is normal.
\end{proof}

\begin{theorem}[Urysohn's Lemma]
Let $E$ and $F$ be disjoint closed subsets of a normal space $X$. Then there exists a continuous function
\[
f:X\to [0,1]
\]
such that
\[
f|_E=0, \qquad f|_F=1.
\]
\end{theorem}

\begin{proof}
Set $V:=X\setminus F$, which is open and contains $E$. Using normality repeatedly, construct for each dyadic rational $r\in(0,1)$ an open set $U_r$ such that
\[
E\subseteq U_r \subseteq V
\]
and
\[
\overline{U_r}\subseteq U_s \qquad \text{whenever } 0<r<s<1.
\]

Define $f:X\to[0,1]$ by
\[
f(x)=
\begin{cases}
0, & x\in \bigcap_{0<r<1} U_r,\\[1mm]
\sup\{r\in(0,1)\cap D : x\notin U_r\}, & \text{otherwise},
\end{cases}
\]
where $D$ denotes the set of dyadic rationals in $(0,1)$.

Then $0\le f\le 1$, and since $E\subseteq U_r$ for every $r$, we have $f=0$ on $E$. Also $U_r\subseteq X\setminus F$ for every $r$, so $f=1$ on $F$.

To prove continuity, let $x\in X$ and $\varepsilon>0$. Choose dyadic rationals $r,s$ such that
\[
f(x)-\varepsilon < r < f(x) < s < f(x)+\varepsilon.
\]
Then $x\notin \overline{U_r}$ and $x\in U_s$. Hence
\[
W:=U_s\setminus \overline{U_r}
\]
is an open neighbourhood of $x$. For every $y\in W$ we have $y\notin U_r$, so $f(y)\ge r$, and $y\in U_s$, so $f(y)\le s$. Thus
\[
r\le f(y)\le s,
\]
and therefore
\[
|f(y)-f(x)|<\varepsilon.
\]
So $f$ is continuous.
\end{proof}

\begin{theorem}[Tietze Extension Theorem]
Let $X$ be a normal topological space, $Y \subseteq X$ closed, and let 
$f:Y \to \mathbb{R}$ be bounded and continuous. Then there exists a bounded continuous function 
$h:X \to \mathbb{R}$ such that $h|_Y = f$.
\end{theorem}

\begin{proof}
Assume $|f|\le c_0$ on $Y$. Using Urysohn's Lemma, construct a continuous function $g_0:X\to\mathbb{R}$ such that
\[
|g_0|\le \tfrac{c_0}{3} \quad \text{on } X, \qquad |f-g_0|\le \tfrac{2c_0}{3} \quad \text{on } Y.
\]
Inductively, having defined $g_0,\dots,g_{n-1}$, set
\[
c_n := \sup_{y\in Y} |f(y)-g_0(y)-\cdots-g_{n-1}(y)|.
\]
Again by Urysohn, construct $g_n$ such that
\[
|g_n|\le \tfrac{c_n}{3} \quad \text{on } X, \qquad 
|f-g_0-\cdots-g_n|\le \tfrac{2c_n}{3} \quad \text{on } Y.
\]
Then $c_n \le (2/3)^n c_0$ and $|g_n|\le (2/3)^n c_0/3$.

The series $\sum g_n$ converges uniformly on $X$ to a continuous function
\[
h := \sum_{n=0}^\infty g_n.
\]
Moreover, on $Y$,
\[
f = \lim_{n\to\infty} (g_0+\cdots+g_n) = h,
\]
and $h$ is bounded. Thus $h$ is the desired extension.
\end{proof}

\subsection*{Basis \& Subbasis}

\begin{definition}
Let $X$ be a set. A collection $\mathcal{B}$ of subsets of $X$ is called a \emph{basis} for a topology on $X$ if:

\begin{enumerate}[label=(\roman*)]
\item For every $x \in X$ there exists $B \in \mathcal{B}$ such that $x \in B$.

\item If $x \in B_1 \cap B_2$ for $B_1,B_2 \in \mathcal{B}$, then there exists
$B_3 \in \mathcal{B}$ such that
\[
x \in B_3 \subseteq B_1 \cap B_2 .
\]
\end{enumerate}
\end{definition}

\begin{theorem}[Topology generated by a basis]
If $\mathcal{B}$ satisfies the two conditions above, then there exists a unique topology $\mathcal{T}$ on $X$ whose open sets are precisely the unions of elements of $\mathcal{B}$. That is,
\[
\mathcal{T}
=
\left\{
\bigcup_{\alpha \in I} B_\alpha
:
B_\alpha \in \mathcal{B},\ I \text{ arbitrary}
\right\}.
\]
\end{theorem}

\begin{definition}
Let $X$ be a set. A collection $\mathcal{S}$ of subsets of $X$ is called a
\emph{subbasis} for a topology on $X$ if the union of all sets in $\mathcal{S}$ equals $X$,
\[
X = \bigcup_{S \in \mathcal{S}} S.
\]

The topology generated by $\mathcal{S}$ is obtained by first forming the basis
consisting of all finite intersections of elements of $\mathcal{S}$,
\[
\mathcal{B}
=
\left\{
S_1 \cap \cdots \cap S_n :
S_i \in \mathcal{S},\ n \in \mathbb{N}
\right\},
\]
and then taking arbitrary unions of these sets.

The topology generated by the subbasis $\mathcal{S}$ consists of all arbitrary unions of elements of this basis. That is, a set $U \subseteq X$ is open if and only if it can be written as
\[
U
=
\bigcup_{\alpha \in I}
\left(
\bigcap_{i=1}^{n_\alpha} S_{\alpha,i}
\right),
\qquad S_{\alpha,i} \in \mathcal{S},\quad n_\alpha \in \mathbb{N}.
\]

\end{definition}

\begin{theorem}[Basis criterion]
A collection $\mathcal{B}$ of subsets of $X$ is a basis for a topology on $X$ if and only if:
\begin{enumerate}[label=(\roman*)]
\item For every $x\in X$, there exists $B\in\mathcal{B}$ such that $x\in B$;
\item If $B_1,B_2\in\mathcal{B}$ and $x\in B_1\cap B_2$, then there exists $B_3\in\mathcal{B}$ such that
\[
x\in B_3 \subseteq B_1\cap B_2.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
$(\Rightarrow)$ If $\mathcal{B}$ is a basis, then (i) holds since $X$ is open, and (ii) holds since $B_1\cap B_2$ is open and can be written as a union of basis elements.

$(\Leftarrow)$ Suppose (i) and (ii) hold. Let $\mathcal{T}$ be the collection of all unions of sets in $\mathcal{B}$. Then $\varnothing, X \in \mathcal{T}$ and arbitrary unions are in $\mathcal{T}$.

To show closure under finite intersections, let $U,V\in\mathcal{T}$ and $x\in U\cap V$. Then there exist $B_1,B_2\in\mathcal{B}$ such that
\[
x\in B_1\subseteq U, \quad x\in B_2\subseteq V.
\]
By (ii), there exists $B_3\in\mathcal{B}$ such that
\[
x\in B_3 \subseteq B_1\cap B_2 \subseteq U\cap V.
\]
Hence
\[
U\cap V = \bigcup_{x\in U\cap V} B_3(x),
\]
so $U\cap V\in\mathcal{T}$. Thus $\mathcal{T}$ is a topology and $\mathcal{B}$ is a basis.
\end{proof}

\begin{theorem}[Lindelöf's Theorem]
Let $X$ be a second-countable topological space. Then every open cover of $X$ admits a countable subcover.
\end{theorem}

\begin{proof}
Let $\mathcal{B} = \{B_n\}_{n\in\mathbb{N}}$ be a countable basis for $X$, and let $\mathcal{U} = \{U_\alpha\}_{\alpha\in A}$ be an open cover of $X$.

For each $n\in\mathbb{N}$, since $B_n \subseteq X = \bigcup_{\alpha\in A} U_\alpha$, we may choose $\alpha(n)\in A$ such that
\[
B_n \subseteq U_{\alpha(n)}.
\]
Then the collection $\{U_{\alpha(n)}\}_{n\in\mathbb{N}}$ is countable.

To see that it covers $X$, let $x\in X$. Since $\mathcal{B}$ is a basis, there exists $n$ such that $x\in B_n$. By construction, $B_n \subseteq U_{\alpha(n)}$, so $x\in U_{\alpha(n)}$. Hence
\[
X = \bigcup_{n=1}^\infty U_{\alpha(n)}.
\]
Thus $\{U_{\alpha(n)}\}_{n\in\mathbb{N}}$ is a countable subcover.
\end{proof}

\subsection*{Compactness}
\begin{theorem}
Let $S$ be a compact subset of a Hausdorff space $X$. For each $x \in X \setminus S$, there exist disjoint open sets $U,V \subseteq X$ such that
\[
x \in U, \qquad S \subseteq V.
\]
\end{theorem}

\begin{proof}
For each $y \in S$, since $X$ is Hausdorff, there exist disjoint open sets $U_y$ and $V_y$ such that
\[
x \in U_y, \qquad y \in V_y.
\]
Then $\{V_y\}_{y\in S}$ is an open cover of $S$. By compactness, there exist $y_1,\dots,y_n \in S$ such that
\[
S \subseteq V_{y_1} \cup \cdots \cup V_{y_n} =: V.
\]
Let
\[
U := U_{y_1} \cap \cdots \cap U_{y_n}.
\]
Then $U$ is open, $x \in U$, and $U \cap V = \varnothing$.
\end{proof}

\begin{corollary}
A compact subset of a Hausdorff space is closed.
\end{corollary}

\begin{proof}
Let $S \subseteq X$ be compact and $x \in X \setminus S$. By the lemma, there exists an open set $U$ containing $x$ with $U \cap S = \varnothing$. Hence $X \setminus S$ is open, so $S$ is closed.
\end{proof}

\begin{theorem}[Theorem 6.5]
Every compact Hausdorff space is normal.
\end{theorem}

\begin{proof}
Let $S,T \subseteq X$ be disjoint closed sets. Since $X$ is compact, $S$ and $T$ are compact. For each $x\in T$, by the lemma on separating a point from a compact set, there exist disjoint open sets $U_x,V_x$ such that
\[
x\in U_x, \qquad S\subseteq V_x.
\]
The family $\{U_x\}_{x\in T}$ is an open cover of $T$, so by compactness there exist $x_1,\dots,x_n\in T$ such that
\[
T\subseteq U_{x_1}\cup\cdots\cup U_{x_n}=:U.
\]
Set
\[
V:=V_{x_1}\cap\cdots\cap V_{x_n}.
\]
Then $U$ and $V$ are open, $T\subseteq U$, $S\subseteq V$, and $U\cap V=\varnothing$. Hence $X$ is normal.
\end{proof}

\begin{theorem}[Theorem 6.6]
Let $f:X\to Y$ be continuous, where $X$ is compact and $Y$ is a topological space. Then $f(X)$ is compact in $Y$.
\end{theorem}

\begin{proof}
Let $\{U_\alpha\}_{\alpha\in A}$ be an open cover of $f(X)$. Then
\[
X=\bigcup_{\alpha\in A} f^{-1}(U_\alpha),
\]
and each $f^{-1}(U_\alpha)$ is open in $X$ by continuity of $f$. Since $X$ is compact, there exist $\alpha_1,\dots,\alpha_n\in A$ such that
\[
X=f^{-1}(U_{\alpha_1})\cup\cdots\cup f^{-1}(U_{\alpha_n}).
\]
Applying $f$, we get
\[
f(X)\subseteq U_{\alpha_1}\cup\cdots\cup U_{\alpha_n}.
\]
Thus $\{U_{\alpha_1},\dots,U_{\alpha_n}\}$ is a finite subcover of $f(X)$, so $f(X)$ is compact.
\end{proof}

\begin{theorem}[Theorem 6.7]
Let $f:X \to Y$ be a continuous map from a compact space $X$ to a Hausdorff space $Y$. If $f$ is one-to-one, then $f$ is a homeomorphism of $X$ onto $f(X)$.
\end{theorem}

\begin{proof}
Since $f$ is continuous and bijective onto $f(X)$, it suffices to show that $f^{-1}:f(X)\to X$ is continuous, or equivalently that $f(U)$ is open in $f(X)$ whenever $U$ is open in $X$.

Let $U\subseteq X$ be open. Then $X\setminus U$ is closed in $X$, hence compact. By continuity, $f(X\setminus U)$ is compact in $Y$, and since $Y$ is Hausdorff, it is closed in $Y$. Therefore it is closed in $f(X)$.

But
\[
f(X\setminus U) = f(X)\setminus f(U)
\]
since $f$ is injective. Hence $f(U)$ is open in $f(X)$, and $f^{-1}$ is continuous. Thus $f$ is a homeomorphism onto $f(X)$.
\end{proof}

\subsubsection*{Arzelà--Ascoli Theorem}

\begin{lemma}
Let $X$ be compact, and let $(f_n)\subset C(X)$ be uniformly bounded and equicontinuous.
Then there exists a subsequence $(f_{n_k})$ which converges uniformly on $X$.
\end{lemma}

\begin{proof}
For each $m\in\mathbb{N}$ and each $x\in X$, equicontinuity gives an open neighbourhood
$U_{x,m}$ such that
\[
|f_n(y)-f_n(x)|<\frac1m
\qquad \text{for all } y\in U_{x,m},\; n\in\mathbb{N}.
\]
By compactness, for each $m$ there exist points $x_{m,1},\dots,x_{m,r_m}$ such that
\[
X=\bigcup_{i=1}^{r_m} U_{x_{m,i},m}.
\]
Let
\[
A := \bigcup_{m=1}^\infty \{x_{m,1}, \dots, x_{m,r_m}\}.
\]
Since this is a countable union of finite sets, we may write
\[
A = \{a_1, a_2, a_3, \dots\}.
\]

Since $(f_n)$ is uniformly bounded, for each $j$ the sequence $(f_n(a_j))$ is bounded,
hence admits a convergent subsequence. We now construct nested subsequences
\[
(f_n)\supset (f_n^{(1)})\supset (f_n^{(2)})\supset (f_n^{(3)})\supset \cdots
\]
such that $(f_n^{(k)}(a_k))$ converges for each $k$.
\[
\begin{aligned}
(f_n^{(1)}) &= (f_{n_1}^{(1)},\, f_{n_2}^{(1)},\, f_{n_3}^{(1)},\, \dots), \\
(f_n^{(2)}) &= (f_{n_1}^{(2)},\, f_{n_2}^{(2)},\, f_{n_3}^{(2)},\, \dots), \\
(f_n^{(3)}) &= (f_{n_1}^{(3)},\, f_{n_2}^{(3)},\, f_{n_3}^{(3)},\, \dots), \\
&\ \vdots
\end{aligned}
\]
where each row is a subsequence of the previous one. \\

Define the diagonal subsequence by $g_k := f_{n_k^{(k)}}.$ Fix~$j$. For all $k\ge j$, the term $g_k$ lies in the subsequence $(f_n^{(j)})$,
hence $(g_k(a_j))$ converges. Thus $(g_k(a_j))$ converges for every $a_j\in A$.

We now show that $(g_k)$ is uniformly Cauchy. Let $\varepsilon>0$ and choose $m$ such that $\frac{3}{m}<\varepsilon.$
From the finite cover
\[
X=\bigcup_{i=1}^{r_m} U_{x_{m,i},m},
\]
and since each $x_{m,i}\in A$, the sequences $(g_k(x_{m,i}))$ converge. Hence there exists $N$
such that for all $k,\ell\ge N$,
\[
|g_k(x_{m,i})-g_\ell(x_{m,i})|<\frac1m
\qquad \text{for } i=1,\dots,r_m.
\]

Now let $x\in X$. Choose $i$ such that $x\in U_{x_{m,i},m}$. Then
\[
\begin{aligned}
|g_k(x)-g_\ell(x)| 
&\le |g_k(x)-g_k(x_{m,i})| + |g_k(x_{m,i})-g_\ell(x_{m,i})| + |g_\ell(x_{m,i})-g_\ell(x)| \\
&< \frac1m+\frac1m+\frac1m = \frac{3}{m} \\
&< \varepsilon
\end{aligned}
\]
Since $x$ arbitrary $(g_k)$ is uniformly Cauchy, hence converges uniformly on $X$.
\end{proof}

\begin{theorem}[Arzelà--Ascoli]
Let $X$ be a compact topological space and let $\mathcal{F} \subset C(X)$.
Then $\overline{\mathcal{F}}$ is compact in $(C(X),\|\cdot\|_\infty)$ $\Leftrightarrow$
$\mathcal{F}$ is uniformly bounded and equicontinuous.
\end{theorem}

\begin{proof}
($\Leftarrow$) Assume $\mathcal{F}$ is uniformly bounded and equicontinuous.
Then by the lemma, every sequence in $\mathcal{F}$ has a uniformly convergent subsequence, hence $\overline{\mathcal{F}}$ sequentially compact.

($\Rightarrow$) Assume $\mathcal{F}$ is precompact. Then $\mathcal{F}$ is totally bounded.

Uniform boundedness:  
For $\varepsilon=1$, there exist $f_1,\dots,f_n$ such that
\[
\mathcal{F} \subset \bigcup_{k=1}^n B(f_k,1).
\]
Hence $\|f\|_\infty \le 1 + \max_k \|f_k\|_\infty$ for all $f \in \mathcal{F}$.

Equicontinuity:  
Let $\varepsilon>0$ and choose $f_1,\dots,f_n$ such that
\[
\mathcal{F} \subset \bigcup_{k=1}^n B(f_k,\varepsilon).
\]
Fix $x\in X$. Since each $f_k$ is continuous, there exists a neighbourhood $U$ of $x$
such that
\[
|f_k(y)-f_k(x)|<\varepsilon \quad \forall y\in U,\; k=1,\dots,n.
\]
If $f\in\mathcal{F}$, choose $k$ with $\|f-f_k\|_\infty<\varepsilon$. Then
\[
|f(y)-f(x)|
\le |f(y)-f_k(y)| + |f_k(y)-f_k(x)| + |f_k(x)-f(x)|
< \varepsilon + \varepsilon + \varepsilon = 3\varepsilon.
\]
\end{proof}

\subsection*{Locally Compact}
\begin{theorem}[One-point compactification]
Let $X$ be a locally compact Hausdorff space. Then there exists a space
$Y = X \cup \{\infty\}$ and a topology on $Y$ such that:

\begin{itemize}
    \item $Y$ is compact and Hausdorff,
    \item the subspace topology on $X$ coincides with the original topology,
    \item the topology is uniquely determined.
\end{itemize}

The open sets in $Y$ are:
\begin{itemize}
    \item all open sets of $X$,
    \item all sets of the form $U \cup \{\infty\}$ where $X \setminus U$ is compact in $X$.
\end{itemize}
\end{theorem}

\subsection*{Connected}

\begin{theorem}[Theorem 8.1]
Let $f$ be a continuous function from a connected topological space $X$ to a topological space $Y$. Then $f(X)$ is connected.
\end{theorem}

\begin{proof}
Let $E \subseteq f(X)$ be both open and closed in $f(X)$. Then $f^{-1}(E)$ is both open and closed in $X$ since $f$ is continuous. 

Because $X$ is connected, we must have either $f^{-1}(E) = \varnothing$ or $f^{-1}(E) = X$. 

In the first case, $E = \varnothing$. In the second case, $E = f(X)$. Hence $f(X)$ is connected.
\end{proof}

\begin{theorem}[Theorem 8.2]
Let $\{E_\alpha\}_{\alpha \in A}$ be a family of connected subsets of a topological space $X$ such that 
$E_\alpha \cap E_\beta \neq \varnothing$ for all $\alpha,\beta$. Then $\bigcup_{\alpha \in A} E_\alpha$ is connected.
\end{theorem}

\begin{proof}
Let $E = \bigcup_\alpha E_\alpha$ and let $F \subseteq E$ be clopen with $F \neq \varnothing$. 
Pick $x \in F \cap E_{\alpha_0}$. Since $E_{\alpha_0}$ is connected, $F \cap E_{\alpha_0} = E_{\alpha_0}$, so $E_{\alpha_0} \subseteq F$.

For any $\beta$, $F \cap E_\beta$ is clopen in $E_\beta$ and contains $E_{\alpha_0} \cap E_\beta \neq \varnothing$, hence $E_\beta \subseteq F$. 
Thus $F = E$.
\end{proof}

\begin{definition}
Let $X$ be a topological space and $x \in X$. The \emph{connected component} of $x$, denoted $C(x)$, is the union of all connected subsets of $X$ that contain $x$.
\end{definition}

\begin{theorem}
$C(x)$ is connected and is the largest connected subset of $X$ containing $x$.
\end{theorem}

\begin{theorem}[Theorem 8.3]
Two connected components of $X$ either coincide or are disjoint. Hence, the connected components form a partition of $X$.
\end{theorem}

\begin{theorem}[Theorem 8.4]
Any interval in $\mathbb{R}$ (closed, open, or half-open; finite or infinite) is connected.
\end{theorem}

\begin{proof}
It suffices to show that $[0,1]$ is connected. Let $E \subseteq [0,1]$ be clopen and assume $E \neq \varnothing$. Set
\[
t = \sup E.
\]
Since $E$ is closed, $t \in E$. If $t < 1$, then since $E$ is open, there exists $\varepsilon > 0$ such that $(t, t+\varepsilon) \subseteq E$, contradicting the definition of $t$. Hence $E = [0,1]$.
\end{proof}

\subsection*{Path Connectedness}
\begin{theorem}[Lemma 9.1]
The relation ``there is a path in $X$ from $x$ to $y$'' is an equivalence relation on $X$.
\end{theorem}

\begin{proof}
We verify the three properties:

\begin{enumerate}
    \item[(i)] \textbf{Reflexive:} For any $x \in X$, the constant map
    \[
    \alpha : [0,1] \to X, \qquad \alpha(s)=x
    \]
    is a path from $x$ to $x$.

    \item[(ii)] \textbf{Symmetric:} If $\alpha$ is a path from $x$ to $y$, then
    \[
    \beta : [0,1] \to X, \qquad \beta(s)=\alpha(1-s)
    \]
    is a path from $y$ to $x$.

    \item[(iii)] \textbf{Transitive:} If $\alpha$ is a path from $x$ to $y$ and $\beta$ is a path from $y$ to $z$, define
    \[
    \gamma(s)=
    \begin{cases}
    \alpha(2s), & 0 \le s \le \tfrac12,\\[1mm]
    \beta(2s-1), & \tfrac12 \le s \le 1.
    \end{cases}
    \]
    Then $\gamma$ is a path from $x$ to $z$.
\end{enumerate}

Hence the relation is an equivalence relation.
\end{proof}

\begin{theorem}[Theorem 9.2]
Every path-connected space is connected.
\end{theorem}

\begin{proof}
Fix $x_0 \in X$. For each $x \in X$, let $\gamma_x:[0,1]\to X$ be a path from $x_0$ to $x$. Since $[0,1]$ is connected and $\gamma_x$ is continuous, $\gamma_x([0,1])$ is connected. Also, each $\gamma_x([0,1])$ contains $x_0$. Hence
\[
X=\bigcup_{x\in X}\gamma_x([0,1])
\]
is a union of connected subsets with a common point, so $X$ is connected.
\end{proof}

\begin{corollary}[Corollory 9.3]
Each connected component of a topological space is a union of path components.
\end{corollary}

\begin{proof}
A path component is path-connected, hence connected by the theorem. Therefore each path component is contained in some connected component. It follows that a connected component is the union of all path components contained in it.
\end{proof}

\subsection*{Finite Product Spaces}

\begin{theorem}[10.1]
Let $X$ be the product of the topological spaces $X_1,\dots,X_n$, and let $\pi_j$ be the projection of $X$ onto $X_j$. The product topology for $X$ is the smallest topology for which each of the projections $\pi_j$ is continuous.
\end{theorem}

\begin{proof}
Let $\mathcal{T}$ be another topology for $X$ such that the projections $\pi_j$ are $\mathcal{T}$-continuous. Let $U_1,\dots,U_n$ be open subsets of $X_1,\dots,X_n$, respectively. Then each $\pi_j^{-1}(U_j)$ belongs to $\mathcal{T}$. Since
\[
\pi_1^{-1}(U_1)\cap\cdots\cap\pi_n^{-1}(U_n)=U_1\times\cdots\times U_n,
\]
the basic set $U_1\times\cdots\times U_n$ belongs to $\mathcal{T}$ and $\mathcal{T}$ includes the product topology.
\end{proof}

\begin{theorem}[10.2]
Let $X=X_1\times\cdots\times X_n$ be the product of the topological spaces $X_1,\dots,X_n$. Then each projection $\pi_j$ is an open map of $X$ onto $X_j$.
\end{theorem}

\begin{proof}
If $U=U_1\times\cdots\times U_n$ is a nonempty basic open set, then $\pi_j(U)=U_j$ is open. Since maps preserve unions, the image of any open set is open.
\end{proof}

\begin{theorem}[10.3]
Let $E$ be a topological space and let $f$ be a function from $E$ to the product
\[
X=X_1\times\cdots\times X_n.
\]
Then $f$ is continuous if and only if each $\pi_j\circ f$ is continuous, $1\le j\le n$.
\end{theorem}

\begin{proof}
If $f$ is continuous, then $\pi_j\circ f$ is also continuous since the composition of continuous functions is continuous.

Conversely, suppose that each $\pi_j\circ f$ is continuous. Let
\[
U=U_1\times\cdots\times U_n
\]
be a basic open set in $X$. Then
\[
f^{-1}(U)=(\pi_1\circ f)^{-1}(U_1)\cap\cdots\cap(\pi_n\circ f)^{-1}(U_n)
\]
is a finite intersection of open sets and hence is open. Since inverses of functions preserve unions, the inverse image of any open set is open. Hence $f$ is continuous.
\end{proof}

\begin{theorem}[10.4]
If $X_1,\dots,X_n$ are Hausdorff spaces, then $X=X_1\times\cdots\times X_n$ is Hausdorff.
\end{theorem}

\begin{proof}
Let $x=(x_1,\dots,x_n)$ and $y=(y_1,\dots,y_n)$ be distinct points of $X$. Choose an index $j$ such that $x_j\ne y_j$. Let $U_j$ and $V_j$ be disjoint open neighborhoods of $x_j$ and $y_j$, respectively, in $X_j$. Then $\pi_j^{-1}(U_j)$ and $\pi_j^{-1}(V_j)$ are disjoint open neighborhoods of $x$ and $y$, respectively, in $X$.
\end{proof}

\begin{theorem}[10.5]
If $X_j$ is a path-connected topological space, $1\le j\le n$, then $X=X_1\times\cdots\times X_n$ is path-connected.
\end{theorem}

\begin{proof}
Let $x=(x_1,\dots,x_n)$ and $y=(y_1,\dots,y_n)$ be points of $X$. For $1\le j\le n$, let
\[
\gamma_j:[0,1]\to X_j
\]
be a path from $x_j$ to $y_j$. Then the path
\[
\gamma:[0,1]\to X,\qquad
\gamma(t)=(\gamma_1(t),\dots,\gamma_n(t)),\qquad 0\le t\le 1,
\]
is a path from $x$ to $y$. Hence $X$ is path-connected.
\end{proof}

\begin{theorem}[10.6]
If $X_j$ is a connected topological space, $1\le j\le n$, then $X=X_1\times\cdots\times X_n$ is also connected.
\end{theorem}

\begin{proof}
Let $E$ be a closed and open subset of $X$. Replacing $E$ by its complement, if necessary, we can assume that $E\ne\varnothing$. We must show that $E=X$.

Let $z=(z_1,\dots,z_n)\in E$ and let $y=(y_1,\dots,y_n)\in X$. The intersection
\[
E\cap (X_1\times\{z_2\}\times\cdots\times\{z_n\})
\]
is a relatively open and closed subset of the coordinate slice, which includes $z$. Since $X_1$ is connected, so is the coordinate slice. It follows that $E$ must include the entire coordinate slice. In particular,
\[
(y_1,z_2,\dots,z_n)\in E.
\]

Now we repeat this argument in the second coordinate and continue until eventually we conclude that $y\in E$. Since $y\in X$ is arbitrary, we obtain $E=X$.
\end{proof}

\begin{lemma}[10.7]
Let $Y$ be a topological space and let $\mathcal{B}$ be a base for the topology of $Y$. If every open cover of $Y$ by sets in $\mathcal{B}$ has a finite subcover, then $Y$ is compact.
\end{lemma}

\begin{proof}
Let $\{U_\beta\}_{\beta\in B}$ be an open cover of $Y$. For each $y\in Y$, choose $V_y\in\mathcal{B}$ and an index $\beta$ such that
\[
y\in V_y\subset U_\beta.
\]
The family $\{V_y:y\in Y\}$ forms an open cover of $Y$ by sets in $\mathcal{B}$. By hypothesis, there exist a finite number of the $V_y$'s that cover $Y$. Since each of these $V_y$'s is contained in at least one of the $U_\beta$'s, we obtain a finite number of the $U_\beta$'s that cover $Y$. Hence $Y$ is compact.
\end{proof}

\begin{theorem}[10.8, Tychonoff's Theorem, finite case]
If $X_1,\dots,X_n$ are compact spaces, then $X_1\times\cdots\times X_n$ is compact.
\end{theorem}

\begin{proof}
It suffices to prove the theorem in the case where $n=2$, and so we consider only
\[
X=X_1\times X_2.
\]
Let $\mathcal{C}$ be a cover of $X_1\times X_2$ by rectangular open sets of the form $U\times V$, where $U$ is open in $X_1$ and $V$ is open in $X_2$. By Lemma 10.7, it suffices to show that the cover $\mathcal{C}$ has a finite subcover.

Fix $z\in X_2$. The coordinate slice $X_1\times\{z\}$ is compact. Hence there are a finite number of sets
\[
U_1\times V_1,\dots,U_m\times V_m\in\mathcal{C}
\]
that cover $X_1\times\{z\}$. We can assume that $z\in V_j$ for all $j$, $1\le j\le m$. Then
\[
V(z)=V_1\cap\cdots\cap V_m
\]
is an open neighborhood of $z$ in $X_2$. Furthermore,
\[
\pi_2^{-1}(V(z))
\]
is covered by a finite number of sets in $\mathcal{C}$, namely the sets
\[
U_j\times V_j,\qquad 1\le j\le m.
\]

The sets $\{V(z):z\in X_2\}$ form an open cover of $X_2$. Since $X_2$ is compact, there exist
\[
z_1,\dots,z_k\in X_2
\]
such that
\[
X_2=V(z_1)\cup\cdots\cup V(z_k).
\]
Then
\[
X=\pi_2^{-1}(V(z_1))\cup\cdots\cup \pi_2^{-1}(V(z_k)).
\]
Since each $\pi_2^{-1}(V(z_j))$ can be covered by a finite number of sets in $\mathcal{C}$, and since the aggregate of these sets covers $X$, $\mathcal{C}$ has a finite subcover.
\end{proof}

\subsection*{Axiom of Choice \& Zorn's Lemma}

\begin{definition}[Axiom of Choice]
Let $\{U_\alpha\}_{\alpha \in A} \subset \mathcal{P}(X)$ be a family of non-empty sets indexed by some set $A$. Then there exists a function $f : A \to X$ such that $f(\alpha) \in U_\alpha$ for all $\alpha \in A$.
\end{definition}

\begin{definition}[Partially Ordered Set]
A set $P$ together with a relation $\leq$ is called a \emph{partially ordered set} if for all $x,y,z \in P$:
\begin{itemize}
    \item $x \leq x$ (reflexivity),
    \item if $x \leq y$ and $y \leq x$, then $x = y$ (antisymmetry),
    \item if $x \leq y$ and $y \leq z$, then $x \leq z$ (transitivity).
\end{itemize}
\end{definition}

\begin{definition}[Totally Ordered Subset]
A subset $C \subset P$ is called \emph{totally ordered} (or a \emph{chain}) if for all $x,y \in C$, either $x \leq y$ or $y \leq x$.
\end{definition}

\begin{definition}[Upper Bound]
Let $C \subset P$. An element $u \in P$ is called an \emph{upper bound} of $C$ if
\[
x \leq u \quad \text{for all } x \in C.
\]
\end{definition}

\begin{definition}[Maximal Element]
An element $m \in P$ is called \emph{maximal} if there is no $x \in P$ such that
\[
m < x.
\]
\end{definition}

\begin{lemma}[Zorn's Lemma]
Let $P$ be a partially ordered set. If each totally ordered subset of $P$ has an upper bound, then $P$ has a maximal element.
\end{lemma}

\subsection*{Infinite Product Spaces}

\begin{theorem}[12.3, Alexander Subbase Theorem]
Let $Y$ be a topological space and let $\mathcal{S}$ be a subbase for the topology of $Y$. If every open cover of $Y$ by sets in $\mathcal{S}$ has a finite subcover, then $Y$ is compact.
\end{theorem}

\begin{proof}
Let $\mathcal{C}$ be an open cover of $Y$ such that no finite subfamily of $\mathcal{C}$ covers $Y$. It suffices to show that there exists a cover of $Y$ by sets in $\mathcal{S}$ that has no finite subcover.

Consider families $\mathcal{B}$ of open subsets of $Y$ such that $\mathcal{C} \subset \mathcal{B}$ and such that no finite collection of sets in $\mathcal{B}$ covers $Y$. The aggregate of such families forms a partially ordered set, with the ordering of inclusion. Let $\{\mathcal{B}_\beta\}_{\beta\in B}$ be a totally ordered subset of $\mathcal{B}$. Let $\mathcal{B}_0$ be the family of subsets of $Y$ that is the union of the $\mathcal{B}_\beta$’s. We claim that $\mathcal{B}_0 \in \mathcal{B}$, that is, that no finite collection of sets in $\mathcal{B}_0$ covers $Y$. Indeed, any finite collection of sets in $\mathcal{B}_0$ is included in one of the $\mathcal{B}_\beta$’s, because of the total ordering, and no finite family of sets in any $\mathcal{B}_\beta$ covers $Y$. (Compare this argument with the corresponding proof that every cover has a basis.) Evidently $\mathcal{B}_0$ is an upper bound for $\{\mathcal{B}_\beta\}$. Hence the hypothesis of Zorn’s Lemma are met. By Zorn’s Lemma, there is a maximal element $\mathcal{B}_0$ in $\mathcal{B}$.

It suffices now to show that the family $\mathcal{S}$ covers $Y$. Indeed, by the defining property of $\mathcal{B}_0$, no finite subfamily of $\mathcal{B}_0$ covers $Y$.

Let $x \in Y$. It suffices to show that there exists $V \in \mathcal{S}$ such that $x \in V$. Choose $U \in \mathcal{B}_0$ such that $x \in U$. Since $\mathcal{S}$ is a subbase, there exist $V_1,\dots,V_n \in \mathcal{S}$ such that
\[
x \in V_1 \cap \cdots \cap V_n \subset U.
\]
Suppose that no $V_j$ belongs to $\mathcal{B}_0$. Because of the maximality of $\mathcal{B}_0$, there exist for each $j$ a finite number of open sets
\[
W_{j1},\dots,W_{jk_j} \in \mathcal{B}_0
\]
such that
\[
W_{j1} \cup \cdots \cup W_{jk_j} \cup V_j
\]
covers $Y$. Otherwise we could add $V_j$ to $\mathcal{B}_0$, contradicting the maximality of $\mathcal{B}_0$. Then
\[
(V_1 \cap \cdots \cap V_n) \cup \bigcup_{j=1}^n \bigcup_{i=1}^{k_j} W_{ji}
\]
covers $Y$. Hence
\[
U \cup \bigcup_{j=1}^n \bigcup_{i=1}^{k_j} W_{ji}
\]
covers $Y$. This gives a finite cover of $Y$ by sets in $\mathcal{B}_0$, contradicting the fact that $\mathcal{B}_0$ belongs to $\mathcal{B}$. We conclude that some $V_j$ belongs to $\mathcal{B}_0$, and we take this $V_j$ to cover $x$.
\end{proof}

\begin{theorem}[12.4, Tychonoff's Theorem]
Any product of compact spaces is compact.
\end{theorem}

\begin{proof}
Let $X=\prod_{\alpha} X_\alpha$, where each $X_\alpha$ is compact. Let $\mathcal{S}$ be a family of subbasic sets of the form
\[
\pi_\alpha^{-1}(U_\alpha),
\]
where $U_\alpha$ is an open subset of $X_\alpha$. Suppose that no finite subfamily of $\mathcal{S}$ covers $X$. In view of the Alexander Subbase Theorem, it suffices to show that $\mathcal{S}$ does not cover $X$.

Fix an index $\beta$ and consider the subbasic sets in $\mathcal{S}$ of the form $\pi_\beta^{-1}(V)$, where $V$ is an open subset of $X_\beta$. The aggregate of such open sets $V$ cannot cover $X_\beta$. Otherwise we could extract a finite subcover $\{V_1,\dots,V_m\}$ of $X_\beta$, and the corresponding sets $\pi_\beta^{-1}(V_1),\dots,\pi_\beta^{-1}(V_m)$ would cover $X$, contradicting our hypothesis on $\mathcal{S}$. Choose a point $x_\beta \in X_\beta$ such that $x_\beta$ is not included in the union of the $V$’s.

The various choices $x_\beta$ determine a point $x \in \prod_\alpha X_\alpha$, which is evidently not included in any of the sets in $\mathcal{S}$.
\end{proof}

\subsection*{Stone--Weierstrass Approximation Theorem}

\begin{definition}[Lattice]
A subset $L \subset C(X,\mathbb{R})$ is called a \emph{lattice} if for all $f,g \in L$, 
\[
\max(f,g) \in L \quad \wedge  \quad \min(f,g) \in L
\]
If $L$ is additionally a subspace (or subalgebra) of $C(X,\mathbb{R})$, it is called a \emph{sublattice}.
\end{definition}

\begin{lemma}
Let $X$ be a compact Hausdorff space with more than one point, and let $L$ be a closed sublattice of $C(X,\mathbb{R})$ with the following property: if $x$ and $y$ are distinct points of $X$ and $a$ and $b$ are two real numbers, then there exists a function $f \in L$ such that $f(x) = a$ and $f(y) = b$. Then $L = C(X,\mathbb{R})$.
\end{lemma}

\begin{lemma}
Let $X$ be an arbitrary topological space. Then every closed subalgebra of $C(X,\mathbb{R})$ is also a closed sublattice of $C(X,\mathbb{R})$.
\end{lemma}

\begin{theorem}[Real Stone--Weierstrass Theorem]
Let $X$ be a compact Hausdorff space, and let $A$ be a closed subalgebra of $C(X,\mathbb{R})$ which separates points and contains a non-zero constant function. Then $A = C(X,\mathbb{R})$.
\end{theorem}

\begin{proof}
If $X$ has only one point, then $C(X,\mathbb{R})$ contains only constant functions; and since $A$ contains a non-zero constant function and is an algebra, it contains all constant functions and equals $C(X,\mathbb{R})$. We may thus assume that $X$ has more than one point.

By the preceding lemmas, it suffices to show that if $x$ and $y$ are distinct points of $X$, and if $a$ and $b$ are any two real numbers, then there exists a function $f \in A$ such that
\[
f(x) = a \quad \text{and} \quad f(y) = b.
\]
Since $A$ separates points, there exists a function $g \in A$ such that $g(x) \neq g(y)$. Define
\[
f(z) = a \frac{g(z) - g(y)}{g(x) - g(y)} + b \frac{g(x) - g(z)}{g(x) - g(y)}.
\]
Then $f \in A$ and clearly satisfies $f(x) = a$ and $f(y) = b$. Hence $A = C(X,\mathbb{R})$.
\end{proof}

\begin{theorem}[Complex Stone--Weierstrass Theorem]
Let $X$ be a compact Hausdorff space, and let $A$ be a closed subalgebra of $C(X,\mathbb{C})$ which separates points, contains a non-zero constant function, and contains the conjugate of each of its functions. Then $A = C(X,\mathbb{C})$.
\end{theorem}

\begin{proof}[Proof outline]
Let $B$ be the set of real-valued functions in $A$. Then $B$ is a closed subalgebra of $C(X,\mathbb{R})$.

If $f \in A$, then its real and imaginary parts satisfy
\[
R(f), I(f) \in C(X,\mathbb{R}),
\]
and since $A$ contains conjugates and is an algebra, one shows that $R(f), I(f) \in B$. Thus $A$ is generated by $B$, and it suffices to show that $B = C(X,\mathbb{R})$.

\begin{itemize}
    \item $B$ separates points (since $A$ does),
    \item $B$ contains a non-zero constant function.
\end{itemize}

By the real Stone--Weierstrass theorem, $B = C(X,\mathbb{R})$, and therefore $A = C(X,\mathbb{C})$.
\end{proof}
